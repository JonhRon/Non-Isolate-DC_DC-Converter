	\section*{Boost Converter Power Stage Calculation (12V to 24V)}
	
	\subsection*{Given Parameters}
	\begin{itemize}
		\item Input Voltage (\(V_{in}\)): 12V  
		\item Output Voltage (\(V_{out}\)): 24V  
		\item Output Power (\(P_{out}\)): 120W  
		\item Switching Frequency (\(F_s\)): 100 kHz  
		\item Ripple Current Rate: 30\%  
		\item Ripple Voltage Rate: 1\%  
	\end{itemize}
	
	\subsection*{1. Duty Cycle (\(D\))}
	For a boost converter in CCM:
	\[
	D = 1 - \frac{V_{in}}{V_{out}} = 1 - \frac{12}{24} = 0.5 \quad (\text{50\%})
	\]
	
	\subsection*{2. Output Current (\(I_{out}\))}
	\[
	I_{out} = \frac{P_{out}}{V_{out}} = \frac{120}{24} = 5 \, \text{A}
	\]
	
	\subsection*{3. Output Resistance (\(R_{out}\))}
	\[
	R_{out} = \frac{V_{out}^2}{P_{out}} = \frac{24^2}{120} = 4.8 \, \Omega
	\]
	
	\subsection*{4. Inductor Current Calculations}
	\begin{itemize}
		\item \textbf{Average Inductor Current (\(I_{L_{avg}}\))}:
		\[
		I_{L_{avg}} = \frac{V_{out} \cdot I_{out}}{V_{in}} = \frac{24 \cdot 5}{12} = 10 \, \text{A}
		\]
		
		\item \textbf{Ripple Current (\(\Delta I_L\))}:
		\[
		\Delta I_L = 0.3 \cdot I_{L_{avg}} = 0.3 \cdot 10 = 3 \, \text{A (peak-to-peak)}
		\]
		
		\item \textbf{Maximum/Minimum Inductor Current}:
		\[
		I_{L_{max}} = 10 + \frac{3}{2} = 11.5 \, \text{A}, \quad 
		I_{L_{min}} = 10 - \frac{3}{2} = 8.5 \, \text{A}
		\]
	\end{itemize}
	
	\subsection*{5. Inductor Value (\(L\))}
	\[
	L = \frac{V_{in} \cdot D}{F_s \cdot \Delta I_L} = \frac{12 \cdot 0.5}{100 \times 10^3 \cdot 3} = 20 \, \mu\text{H}
	\]
	
	\subsection*{6. Output Capacitor (\(C\))}
	\begin{itemize}
		\item \textbf{Ripple Voltage (\(\Delta V_{out}\))}:
		\[
		\Delta V_{out} = 0.01 \cdot V_{out} = 0.24 \, \text{V}
		\]
		
		\item \textbf{Capacitor Value}:
		\[
		C = \frac{I_{out} \cdot D}{F_s \cdot \Delta V_{out}} = \frac{5 \cdot 0.5}{100 \times 10^3 \cdot 0.24} \approx 104 \, \mu\text{F}
		\]
	\end{itemize}
	
	\subsection*{7. MOSFET Selection}
	\begin{itemize}
		\item Voltage Rating: \(\geq 24 \, \text{V}\) (Choose 30V for margin)
		\item Current Rating: \(\geq 10 \, \text{A}\) (Choose 15A for margin)
		\item Example: \textbf{IRF540N} (100V, 33A)
	\end{itemize}
	
	\subsection*{8. Diode Selection}
	\begin{itemize}
		\item Reverse Voltage: \(\geq 24 \, \text{V}\) (Choose 30V Schottky diode)
		\item Average Current: \(\geq 5 \, \text{A}\)
		\item Example: \textbf{STPS20H100CT} (100V, 20A)
	\end{itemize}
